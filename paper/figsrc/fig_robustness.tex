% Robustness of the four controllers for N(2.2, 2.2, pi/2) on the dense noise grid.

\documentclass[border=2pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{amsmath,amssymb}
\usepackage{pgfplots}
\usepackage{pgfplotstable}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc}

\newcommand{\datadir}{../../data}
\newcommand{\rcsv}{\datadir/robustness_analysis_3/combined_ewma_data.csv}

\definecolor{cAdam}{RGB}{31,78,160}
\definecolor{cNom}{RGB}{224,109,30}
\definecolor{cNoise}{RGB}{20,120,116}

\pgfplotsset{
  a70/.style={color=cAdam, thick},
  a60/.style={color=cAdam!55!white, thick, dashed},
  nom/.style={color=cNom, thick},
  noi/.style={color=cNoise, thick},
}

\begin{document}
\begin{tikzpicture}

% ------------------------------------------------------------------ panel (a)
\begin{axis}[
    name=mainplot,
    width=8.2cm, height=6.4cm,
    xlabel={Noise strength $\sigma_{\mathrm{noise}}$ (MHz)},
    ylabel={Average fidelity $\overline{F}$},

    ymin=0.62, ymax=1.01,
    grid=major,
    grid style={dashed, gray!30},
    label style={font=\small},
    tick label style={font=\footnotesize},
    legend to name=roblegend, legend columns=4,
    legend style={font=\scriptsize, legend cell align=left, draw=none,
                  /tikz/every even column/.append style={column sep=9pt}},

    filter discard warning=false,
    unbounded coords=discard
]
    \addplot[color=cAdam,  opacity=0.12, forget plot] table[x=sigma_mhz, y=adam70_fidelity_raw,  col sep=comma] {\rcsv};
    \addplot[color=cNom,   opacity=0.12, forget plot] table[x=sigma_mhz, y=nominal_fidelity_raw, col sep=comma] {\rcsv};
    \addplot[color=cNoise, opacity=0.12, forget plot] table[x=sigma_mhz, y=noise_fidelity_raw,   col sep=comma] {\rcsv};

          fill opacity=0.85, text opacity=1, inner sep=1pt]
        at (axis cs:1.0,0.885) {$\sigma_{\mathrm{train}}$};

    \addplot[a70] table[x=sigma_mhz, y=adam70_fidelity_ewma,  col sep=comma] {\rcsv};
    \addlegendentry{direct optimizer, $70\,$ns}
    \addplot[a60] table[x=sigma_mhz, y=adam60_fidelity_ewma,  col sep=comma] {\rcsv};
    \addlegendentry{direct optimizer, $60\,$ns}
    \addplot[noi] table[x=sigma_mhz, y=noise_fidelity_ewma,   col sep=comma] {\rcsv};
    \addlegendentry{RL, noise-trained, $98\,$ns}
    \addplot[nom] table[x=sigma_mhz, y=nominal_fidelity_ewma, col sep=comma] {\rcsv};
    \addlegendentry{RL, noise-free, $116\,$ns}
\end{axis}

% inset: low-noise magnification
\begin{axis}[
