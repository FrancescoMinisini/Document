% Generic standalone wrapper for regenerating a thesis figure PDF.
% Compile FROM THE REPOSITORY ROOT so that the data/... paths inside the
% plot sources resolve, e.g.:
%
%   pdflatex -jobname=avg_fidelity_vs_noise \
%       "\def\PlotFile{plots/avg_fidelity_vs_noise.tex}\input{plots/_standalone_wrapper.tex}"
%
% varwidth fixes \linewidth to the thesis text width (report, a4paper,
% margin=0.9in  ->  ~466pt), so figures come out at the same size as when
% they are \input directly into Tesi.tex.
\documentclass[border=2pt,varwidth=466pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{amsmath,amssymb,bm}
\usepackage{braket}
\usepackage{pgfplots}
\usepackage{pgfplotstable}
\pgfplotsset{compat=1.18}
\usepgfplotslibrary{statistics,groupplots}
\usepackage{tikz}
\usetikzlibrary{arrows.meta,positioning,calc,fit,backgrounds,decorations.pathreplacing}

\begin{document}
\input{\PlotFile}
\end{document}
